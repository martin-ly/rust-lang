% 中文支持（LuaLaTeX）
\usepackage[fontset=windows]{ctex}

% 页面边距
\usepackage{geometry}
\geometry{a4paper, margin=2.5cm}

% 扩展嵌套列表深度（默认 4 层不够）
\usepackage{enumitem}
\setlistdepth{10}
\renewlist{itemize}{itemize}{10}
\renewlist{enumerate}{enumerate}{10}
\setlist[itemize]{label=$\cdot$}
\setlist[itemize,1]{label=\textbullet}
\setlist[itemize,2]{label=--}
\setlist[itemize,3]{label=*}
\setlist[enumerate,1]{label=\arabic*.}
\setlist[enumerate,2]{label=(\alph*)}
\setlist[enumerate,3]{label=(\roman*)}

% 代码块换行
\usepackage{fvextra}
\DefineVerbatimEnvironment{Highlighting}{Verbatim}{breaklines,commandchars=\\\{\}}

% 链接颜色
\usepackage{xcolor}
\usepackage{hyperref}
\definecolor{linkblue}{RGB}{0,51,102}
\hypersetup{colorlinks=true,linkcolor=linkblue,urlcolor=linkblue,citecolor=linkblue}

% 用 ulem 的 \sout 覆盖 soul 的 \st，避免 verbatim/CJK 混排时 Reconstruction failed
\usepackage{ulem}
\AtBeginDocument{\renewcommand{\st}[1]{\sout{#1}}}
